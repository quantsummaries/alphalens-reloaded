  36def customize(func):
  37    """
  38    Decorator to set plotting context and axes style during function call.
  39    """
  40
  41    @wraps(func)
  42    def call_w_context(*args, **kwargs):
  43        set_context = kwargs.pop("set_context", True)
  44        if set_context:
  45            color_palette = sns.color_palette("colorblind")
  46            with warnings.catch_warnings():
  47                warnings.filterwarnings(
  48                    "ignore",
  49                    message="The locs attribute was deprecated in Matplotlib 3.11.*",
  50                    category=MatplotlibDeprecationWarning,
  51                )
  52                warnings.filterwarnings(
  53                    "ignore",
  54                    message="The set_bad function will be deprecated.*",
  55                    category=PendingDeprecationWarning,
  56                )
  57                with plotting_context(), axes_style(), color_palette:
  58                    sns.despine(left=True)
  59                    return func(*args, **kwargs)
  60        else:
  61            return func(*args, **kwargs)
  62
  63    return call_w_context

  66def plotting_context(context="notebook", font_scale=1.5, rc=None):
  67    """
  68    Create alphalens default plotting style context.
  69
  70    Under the hood, calls and returns seaborn.plotting_context() with
  71    some custom settings. Usually you would use in a with-context.
  72
  73    Parameters
  74    ----------
  75    context : str, optional
  76        Name of seaborn context.
  77    font_scale : float, optional
  78        Scale font by factor font_scale.
  79    rc : dict, optional
  80        Config flags.
  81        By default, {'lines.linewidth': 1.5}
  82        is being used and will be added to any
  83        rc passed in, unless explicitly overriden.
  84
  85    Returns
  86    -------
  87    seaborn plotting context
  88
  89    Example
  90    -------
  91    with alphalens.plotting.plotting_context(font_scale=2):
  92        alphalens.create_full_tear_sheet(..., set_context=False)
  93
  94    See also
  95    --------
  96    For more information, see seaborn.plotting_context().
  97    """
  98    if rc is None:
  99        rc = {}
 100
 101    rc_default = {"lines.linewidth": 1.5}
 102
 103    # Add defaults if they do not exist
 104    for name, val in rc_default.items():
 105        rc.setdefault(name, val)
 106
 107    return sns.plotting_context(context=context, font_scale=font_scale, rc=rc)

 149def _plot_summary_table(summary, title, ax=None):
 150    """Render a summary table on a supplied axis or a new figure."""
 151    if ax is None:
 152        _, ax = plt.subplots(
 153            figsize=(max(8, 5 + len(summary.columns)), 1 + 0.45 * len(summary))
 154        )
 155    ax.axis("off")
 156    table = ax.table(
 157        cellText=[[f"{value:.3f}" for value in row] for row in summary.to_numpy()],
 158        colLabels=summary.columns,
 159        rowLabels=summary.index,
 160        loc="center",
 161        cellLoc="center",
 162    )
 163    table.auto_set_font_size(False)
 164    table.set_fontsize(9)
 165    table.scale(1, 1.5)
 166    ax.set_title(title, fontweight="bold", pad=10)
 167    return ax

 220def plot_information_table(
 221    ic_data, return_df=False, tables_as_figures=True, ic_type="spearman"
 222):
 223    """Display IC statistics as a figure or text; always return the DataFrame."""
 224
 225    def _safe_ttest_1samp(series):
 226        clean = series.dropna()
 227        if clean.size < 2 or np.isclose(clean.std(ddof=0), 0.0):
 228            return np.nan, np.nan
 229        with warnings.catch_warnings():
 230            warnings.simplefilter("ignore", RuntimeWarning)
 231            t_stat, p_value = stats.ttest_1samp(clean.values, 0)
 232        return t_stat, p_value
 233
 234    def _safe_moment(series, moment_fn):
 235        clean = series.dropna()
 236        if clean.size < 3 or np.isclose(clean.std(ddof=0), 0.0):
 237            return np.nan
 238        with warnings.catch_warnings():
 239            warnings.simplefilter("ignore", RuntimeWarning)
 240            return moment_fn(clean.values)
 241
 242    ic_summary_table = pd.DataFrame()
 243    ic_summary_table[f"IC Mean ({ic_type})"] = ic_data.mean()
 244    ic_summary_table["IC Std."] = ic_data.std()
 245    ic_summary_table["Risk-Adjusted IC (ICIR)"] = ic_data.mean() / ic_data.std()
 246    t_stats = {}
 247    p_values = {}
 248    for col in ic_data.columns:
 249        t_stat, p_value = _safe_ttest_1samp(ic_data[col])
 250        t_stats[col] = t_stat
 251        p_values[col] = p_value
 252
 253    ic_summary_table["t-stat(IC)"] = pd.Series(t_stats)
 254    ic_summary_table["p-value(IC)"] = pd.Series(p_values)
 255    ic_summary_table["IC Skew"] = ic_data.apply(
 256        lambda col: _safe_moment(col, stats.skew), axis=0
 257    )
 258    ic_summary_table["IC Kurtosis"] = ic_data.apply(
 259        lambda col: _safe_moment(col, stats.kurtosis), axis=0
 260    )
 261
 262    # Ljung-Box test for autocorrelation and Newey-West adjusted t-statistic for IC
 263    ic_supplements_df = utils.ic_autocor_adj(ic_data, lag=1)
 264    ic_summary_table = pd.concat([ic_summary_table, ic_supplements_df.T], axis=1)
 265
 266    if return_df:
 267        return ic_summary_table
 268
 269    # Render a Matplotlib table when requested, but keep data return type stable.
 270    if tables_as_figures:
 271        fig, ax = plt.subplots(figsize=(8, 3))
 272        ax.axis("off")
 273        ax.axis("tight")
 274        table_data = ic_summary_table.T.map(lambda x: f"{x:.3f}")
 275
 276        table = ax.table(
 277            cellText=table_data.values,
 278            colLabels=table_data.columns,
 279            rowLabels=table_data.index,
 280            loc="center",
 281            cellLoc="center"
 282        )
 283        table.auto_set_font_size(False)
 284        table.set_fontsize(9)
 285        table.scale(1.2, 1.2)
 286        ax.set_title(f"Information Analysis Summary ({ic_type})", fontweight="bold", pad=10)
 287
 288    if not tables_as_figures:
 289        print(f"Information Analysis ({ic_type})")
 290        utils.print_table(ic_summary_table.apply(lambda x: x.round(3)).T)
 291
 292    return ic_summary_table

 310def plot_ic_ts(ic, ax=None):
 311    """
 312    Plots Spearman Rank Information Coefficient and IC moving
 313    average for a given factor.
 314
 315    Parameters
 316    ----------
 317    ic : pd.DataFrame
 318        DataFrame indexed by date, with IC for each forward return.
 319    ax : matplotlib.Axes, optional
 320        Axes upon which to plot.
 321
 322    Returns
 323    -------
 324    ax : matplotlib.Axes
 325        The axes that were plotted on.
 326    """
 327    ic = ic.copy()
 328
 329    num_plots = len(ic.columns)
 330    if ax is None:
 331        f, ax = plt.subplots(num_plots, 1, figsize=(18, num_plots * 7))
 332        ax = np.asarray([ax]).flatten()
 333
 334    ymin, ymax = (None, None)
 335    for a, (period_num, ic) in zip(ax, ic.items()):
 336        ic.plot(alpha=0.7, ax=a, lw=0.7, color="steelblue")
 337        ic.rolling(window=22).mean().plot(ax=a, color="forestgreen", lw=2, alpha=0.8)
 338
 339        a.set(ylabel="IC", xlabel="")
 340        a.set_title(
 341            "{} Period Forward Return Information Coefficient (IC)".format(period_num)
 342        )
 343        a.axhline(0.0, linestyle="-", color="black", lw=1, alpha=0.8)
 344        a.legend(["IC", "1 month moving avg"], loc="upper right")
 345        a.text(
 346            0.05,
 347            0.95,
 348            "Mean %.3f \n Std. %.3f" % (ic.mean(), ic.std()),
 349            fontsize=16,
 350            bbox={"facecolor": "white", "alpha": 1, "pad": 5},
 351            transform=a.transAxes,
 352            verticalalignment="top",
 353        )
 354
 355        curr_ymin, curr_ymax = a.get_ylim()
 356        ymin = curr_ymin if ymin is None else min(ymin, curr_ymin)
 357        ymax = curr_ymax if ymax is None else max(ymax, curr_ymax)
 358
 359    for a in ax:
 360        a.set_ylim([ymin, ymax])
 361
 362    return ax

 365def plot_ic_hist(ic, ax=None, plot_kws=None):
 366    """
 367    Plots Spearman Rank Information Coefficient histogram for a given factor.
 368
 369    Parameters
 370    ----------
 371    ic : pd.DataFrame
 372        DataFrame indexed by date, with IC for each forward return.
 373    ax : matplotlib.Axes, optional
 374        Axes upon which to plot.
 375    plot_kws : dict, optional
 376        Additional keyword arguments passed to sns.histplot. E.g., plot_kws={"bins": 20}.
 377
 378    Returns
 379    -------
 380    ax : matplotlib.Axes
 381        The axes that were plotted on.
 382    """
 383    if plot_kws is None:
 384        plot_kws = {}
 385
 386    ic = ic.copy()
 387
 388    num_plots = len(ic.columns)
 389
 390    v_spaces = ((num_plots - 1) // 3) + 1
 391
 392    if ax is None:
 393        f, ax = plt.subplots(v_spaces, 3, figsize=(18, v_spaces * 6))
 394        ax = ax.flatten()
 395
 396    for a, (period_num, ic) in zip(ax, ic.items()):
 397        clean_ic = ic.dropna()
 398
 399        has_variance = clean_ic.nunique() > 1
 400
 401        default_kws = {
 402            'bins': 20,
 403            'binrange': (-1, 1),
 404            'kde': True if has_variance else False
 405        }
 406
 407        merged_kws = {**default_kws, **plot_kws}
 408
 409        sns.histplot(clean_ic, ax=a, **merged_kws)
 410        a.set(title="%s Period IC" % period_num, xlabel="IC")
 411        a.set_xlim([-1, 1])
 412        a.text(
 413            0.05,
 414            0.95,
 415            "Mean %.3f \n Std. %.3f" % (clean_ic.mean(), clean_ic.std()),
 416            fontsize=16,
 417            bbox={"facecolor": "white", "alpha": 1, "pad": 5},
 418            transform=a.transAxes,
 419            verticalalignment="top",
 420        )
 421        a.axvline(clean_ic.mean(), color="w", linestyle="dashed", linewidth=2)
 422
 423    if num_plots < len(ax):
 424        ax[-1].set_visible(False)
 425
 426    return ax

 429def plot_ic_qq(ic, theoretical_dist=stats.norm, ax=None):
 430    """
 431    Plots Spearman Rank Information Coefficient "Q-Q" plot relative to
 432    a theoretical distribution.
 433
 434    Parameters
 435    ----------
 436    ic : pd.DataFrame
 437        DataFrame indexed by date, with IC for each forward return.
 438    theoretical_dist : scipy.stats._continuous_distns
 439        Continuous distribution generator. scipy.stats.norm and
 440        scipy.stats.t are popular options.
 441    ax : matplotlib.Axes, optional
 442        Axes upon which to plot.
 443
 444    Returns
 445    -------
 446    ax : matplotlib.Axes
 447        The axes that were plotted on.
 448    """
 449
 450    ic = ic.copy()
 451
 452    num_plots = len(ic.columns)
 453
 454    v_spaces = ((num_plots - 1) // 3) + 1
 455
 456    if ax is None:
 457        f, ax = plt.subplots(v_spaces, 3, figsize=(18, v_spaces * 6))
 458        ax = ax.flatten()
 459
 460    if isinstance(theoretical_dist, stats.norm.__class__):
 461        dist_name = "Normal"
 462    elif isinstance(theoretical_dist, stats.t.__class__):
 463        dist_name = "T"
 464    else:
 465        dist_name = "Theoretical"
 466
 467    for a, (period_num, ic) in zip(ax, ic.items()):
 468        sm.qqplot(
 469            ic.dropna().values,
 470            theoretical_dist,
 471            fit=True,
 472            line="45",
 473            ax=a,
 474        )
 475        a.set(
 476            title="{} Period IC {} Dist. Q-Q".format(period_num, dist_name),
 477            ylabel="Observed Quantile",
 478            xlabel="{} Distribution Quantile".format(dist_name),
 479        )
 480
 481    return ax

 484def plot_quantile_returns_bar(
 485    mean_ret_by_q, by_group=False, ylim_percentiles=None, ax=None
 486):
 487    """
 488    Plots mean period wise returns for factor quantiles.
 489
 490    Parameters
 491    ----------
 492    mean_ret_by_q : pd.DataFrame
 493        DataFrame with quantile, (group) and mean period wise return values.
 494    by_group : bool
 495        Disaggregated figures by group.
 496    ylim_percentiles : tuple of integers
 497        Percentiles of observed data to use as y limits for plot.
 498    ax : matplotlib.Axes, optional
 499        Axes upon which to plot.
 500
 501    Returns
 502    -------
 503    ax : matplotlib.Axes
 504        The axes that were plotted on.
 505    """
 506
 507    mean_ret_by_q = mean_ret_by_q.copy()
 508
 509    if ylim_percentiles is not None:
 510        ymin = (
 511            np.nanpercentile(mean_ret_by_q.values, ylim_percentiles[0]) * DECIMAL_TO_BPS
 512        )
 513        ymax = (
 514            np.nanpercentile(mean_ret_by_q.values, ylim_percentiles[1]) * DECIMAL_TO_BPS
 515        )
 516    else:
 517        ymin = None
 518        ymax = None
 519
 520    if by_group:
 521        num_group = len(mean_ret_by_q.index.get_level_values("group").unique())
 522
 523        if ax is None:
 524            v_spaces = ((num_group - 1) // 2) + 1
 525            f, ax = plt.subplots(
 526                v_spaces,
 527                2,
 528                sharex=False,
 529                sharey=True,
 530                figsize=(18, 6 * v_spaces),
 531            )
 532            ax = ax.flatten()
 533
 534        for a, (sc, cor) in zip(ax, mean_ret_by_q.groupby(level="group", observed=True)):
 535            (
 536                cor.xs(sc, level="group")
 537                .multiply(DECIMAL_TO_BPS)
 538                .plot(kind="bar", title=sc, ax=a)
 539            )
 540
 541            a.set(xlabel="", ylabel="Mean Return (bps)", ylim=(ymin, ymax))
 542
 543        if num_group < len(ax):
 544            ax[-1].set_visible(False)
 545
 546        return ax
 547
 548    else:
 549        if ax is None:
 550            f, ax = plt.subplots(1, 1, figsize=(18, 6))
 551
 552        (
 553            mean_ret_by_q.multiply(DECIMAL_TO_BPS).plot(
 554                kind="bar",
 555                title="Mean Period Wise Return By Factor Quantile",
 556                ax=ax,
 557            )
 558        )
 559        ax.set(xlabel="", ylabel="Mean Return (bps)", ylim=(ymin, ymax))
 560
 561        return ax

 631def plot_mean_quantile_returns_spread_time_series(
 632    mean_returns_spread, std_err=None, bandwidth=1, ax=None
 633):
 634    """
 635    Plots mean period wise returns for factor quantiles.
 636
 637    Parameters
 638    ----------
 639    mean_returns_spread : pd.Series
 640        Series with difference between quantile mean returns by period.
 641    std_err : pd.Series
 642        Series with standard error of difference between quantile
 643        mean returns each period.
 644    bandwidth : float
 645        Width of displayed error bands in standard deviations.
 646    ax : matplotlib.Axes, optional
 647        Axes upon which to plot.
 648
 649    Returns
 650    -------
 651    ax : matplotlib.Axes
 652        The axes that were plotted on.
 653    """
 654
 655    if isinstance(mean_returns_spread, pd.DataFrame):
 656        if ax is None:
 657            ax = [None for a in mean_returns_spread.columns]
 658
 659        ymin, ymax = (None, None)
 660        for (i, a), (name, fr_column) in zip(
 661            enumerate(ax), mean_returns_spread.items()
 662        ):
 663            stdn = None if std_err is None else std_err[name]
 664            a = plot_mean_quantile_returns_spread_time_series(
 665                fr_column, std_err=stdn, bandwidth=bandwidth, ax=a
 666            )
 667            ax[i] = a
