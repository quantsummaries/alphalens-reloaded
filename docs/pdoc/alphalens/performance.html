  29def _reindex_to_freq(result, freq):
  30    """Attach frequency metadata without changing observations."""
  31    if freq is None:
  32        return result
  33    try:
  34        new_index = pd.DatetimeIndex(result.index, freq=freq, copy=True)
  35    except ValueError:
  36        return result
  37    reindexed = result.copy()
  38    reindexed.index = new_index
  39    return reindexed

  42def factor_information_coefficient(factor_data, group_adjust=False, by_group=False,
  43                                   ic_type="spearman"):
  44    """
  45    Computes the Spearman Rank Correlation or Pearson Correlation based Information Coefficient (IC)
  46    between factor values and N period forward returns for each period in the factor index.
  47
  48    Parameters
  49    ----------
  50    factor_data : pd.DataFrame - MultiIndex
  51        A MultiIndex DataFrame indexed by date (level 0) and asset (level 1),
  52        containing the values for a single alpha factor, forward returns for
  53        each period, the factor quantile/bin that factor value belongs to, and
  54        (optionally) the group the asset belongs to.
  55        - See full explanation in utils.get_clean_factor_and_forward_returns
  56    group_adjust : bool
  57        Demean forward returns by group before computing IC, so that the result reflects factor skill beyond group-level
  58        effects and ICs across groups can be compared. Defaults to False as forward returns are often residual returns,
  59        which are already group-neutral.
  60    by_group : bool
  61        Whether to compute IC statistics separately for each group instead of only one pooled summary
  62        across all assets. It does not change the IC computation method; it changes how results are broken out and
  63        displayed. Defaults to False unless the factor data has the 'group' column.
  64    ic_type : str
  65        Type of information coefficient to compute ("spearman" or "pearson").
  66
  67    Returns
  68    -------
  69    ic : pd.DataFrame
  70        Spearman Rank correlation or Pearson correlation between factor and
  71        provided forward returns; indexed by 'date' with columns like '1D', '2D', etc..
  72    """
  73
  74    def _safe_corr(x, f, method):
  75        valid = pd.concat([x, f], axis=1).dropna()
  76        if len(valid) < 2:
  77            return np.nan
  78        x_valid = valid.iloc[:, 0]
  79        f_valid = valid.iloc[:, 1]
  80        if x_valid.nunique() <= 1 or f_valid.nunique() <= 1:
  81            return np.nan
  82        if method == "spearman":
  83            return stats.spearmanr(x_valid, f_valid)[0]
  84        return stats.pearsonr(x_valid, f_valid)[0]
  85
  86    def src_ic(group):
  87        f = group["factor"]
  88        _ic = group[utils.get_forward_returns_columns(factor_data.columns)].apply(
  89            lambda x: _safe_corr(x, f, "spearman")
  90        )
  91        return _ic
  92
  93    def src_ic_pearson(group):
  94        f = group["factor"]
  95        _ic = group[utils.get_forward_returns_columns(factor_data.columns)].apply(
  96            lambda x: _safe_corr(x, f, "pearson")
  97        )
  98        return _ic
  99
 100    date_idx = factor_data.index.names.index("date")
 101    freq = getattr(factor_data.index.levels[date_idx], "freq", None)
 102
 103    factor_data = factor_data.copy()
 104
 105    grouper = [factor_data.index.get_level_values("date")]
 106
 107    if group_adjust:
 108        factor_data = utils.demean_forward_returns(factor_data, grouper + ["group"])
 109    if by_group:
 110        grouper.append("group")
 111
 112    if ic_type == "spearman":
 113        ic = factor_data.groupby(grouper, observed=True).apply(src_ic, include_groups=False)
 114    elif ic_type == "pearson":
 115        ic = factor_data.groupby(grouper, observed=True).apply(src_ic_pearson, include_groups=False)
 116    else:
 117        raise ValueError("Invalid information coefficient type. Use 'spearman' or 'pearson'.")
 118
 119    if by_group:
 120        return ic
 121    else:
 122        return _reindex_to_freq(ic, freq)

 125def mean_information_coefficient(
 126    factor_data, group_adjust=False, by_group=False, by_time=None
 127):
 128    """
 129    Get the mean information coefficient of specified groups.
 130    Answers questions like:
 131    What is the mean IC for each month?
 132    What is the mean IC for each group for our whole timerange?
 133    What is the mean IC for each group, each week?
 134
 135    Parameters
 136    ----------
 137    factor_data : pd.DataFrame - MultiIndex
 138        A MultiIndex DataFrame indexed by date (level 0) and asset (level 1),
 139        containing the values for a single alpha factor, forward returns for
 140        each period, the factor quantile/bin that factor value belongs to, and
 141        (optionally) the group the asset belongs to.
 142        - See full explanation in utils.get_clean_factor_and_forward_returns
 143    group_adjust : bool
 144        Demean forward returns by group before computing IC.
 145    by_group : bool
 146        If True, take the mean IC for each group.
 147    by_time : str (pd time_rule), optional
 148        Time window to use when taking mean IC.
 149        See http://pandas.pydata.org/pandas-docs/stable/timeseries.html
 150        for available options.
 151
 152    Returns
 153    -------
 154    ic : pd.DataFrame
 155        Mean Spearman Rank correlation between factor and provided
 156        forward price movement windows.
 157    """
 158
 159    ic = factor_information_coefficient(factor_data, group_adjust, by_group)
 160
 161    grouper = []
 162    if by_time is not None:
 163        grouper.append(pd.Grouper(freq=by_time))
 164    if by_group:
 165        grouper.append("group")
 166
 167    if len(grouper) == 0:
 168        ic = ic.mean()
 169
 170    else:
 171        with warnings.catch_warnings():
 172            warnings.filterwarnings(
 173                "ignore",
 174                message="The 'generic' unit for NumPy timedelta is deprecated.*",
 175                category=DeprecationWarning,
 176            )
 177            ic = (
 178                ic.reset_index()
 179                .set_index("date")
 180                .groupby(grouper, observed=False)
 181                .mean()
 182            )
 183
 184    return ic

 187def factor_weights(factor_data, demeaned=True, group_adjust=False, equal_weight=False):
 188    """
 189    Computes asset weights by factor values and dividing by the sum of their
 190    absolute value (achieving gross leverage of 1). Positive factor values will
 191    result in positive weights and negative values in negative weights.
 192
 193    Parameters
 194    ----------
 195    factor_data : pd.DataFrame - MultiIndex
 196        A MultiIndex DataFrame indexed by date (level 0) and asset (level 1),
 197        containing the values for a single alpha factor, forward returns for
 198        each period, the factor quantile/bin that factor value belongs to, and
 199        (optionally) the group the asset belongs to.
 200        - See full explanation in utils.get_clean_factor_and_forward_returns
 201    demeaned : bool
 202        Should this computation happen on a long short portfolio? if True,
 203        weights are computed by demeaning factor values and dividing by the sum
 204        of their absolute value (achieving gross leverage of 1). The sum of
 205        positive weights will be the same as the negative weights (absolute
 206        value), suitable for a dollar neutral long-short portfolio. Defaults to True.
 207    group_adjust : bool
 208        Should this computation happen on a group neutral portfolio? If True,
 209        compute group neutral weights: each group will weight the same and
 210        if 'demeaned' is enabled the factor values demeaning will occur on the
 211        group level.
 212    equal_weight : bool, optional
 213        if True the assets will be equal-weighted instead of factor-weighted
 214        If demeaned is True then the factor universe will be split in two
 215        equal sized groups, top assets with positive weights and bottom assets
 216        with negative weights
 217
 218    Returns
 219    -------
 220    returns : pd.Series
 221        Assets weighted by factor value.
 222    """
 223
 224    def to_weights(group, _demeaned, _equal_weight):
 225
 226        if _equal_weight:
 227            group = group.copy()
 228
 229            if _demeaned:
 230                # top assets positive weights, bottom ones negative
 231                group = group - group.median()
 232
 233            negative_mask = group < 0
 234            group[negative_mask] = -1.0
 235            positive_mask = group > 0
 236            group[positive_mask] = 1.0
 237
 238            if _demeaned:
 239                # positive weights must equal negative weights
 240                if negative_mask.any():
 241                    group[negative_mask] /= negative_mask.sum()
 242                if positive_mask.any():
 243                    group[positive_mask] /= positive_mask.sum()
 244
 245        elif _demeaned:
 246            group = group - group.mean()
 247
 248        gross = group.abs().sum()
 249        if gross == 0 or pd.isna(gross):
 250            return group * 0.0
 251        return group / gross
 252
 253    grouper = [factor_data.index.get_level_values("date")]
 254    if group_adjust:
 255        grouper.append("group")
 256
 257    weights = factor_data.groupby(grouper, group_keys=False, observed=True)[
 258        "factor"
 259    ].apply(to_weights, demeaned, equal_weight)
 260
 261    if group_adjust:
 262        weights = weights.groupby(level="date", group_keys=False).apply(
 263            to_weights, False, False
 264        )
 265
 266    return weights

 269def factor_returns(
 270    factor_data,
 271    demeaned=True,
 272    group_adjust=False,
 273    equal_weight=False,
 274    by_asset=False,
 275):
 276    """
 277    Computes period wise returns for portfolio weighted by factor
 278    values.
 279
 280    Parameters
 281    ----------
 282    factor_data : pd.DataFrame - MultiIndex
 283        A MultiIndex DataFrame indexed by date (level 0) and asset (level 1),
 284        containing the values for a single alpha factor, forward returns for
 285        each period, the factor quantile/bin that factor value belongs to, and
 286        (optionally) the group the asset belongs to.
 287        - See full explanation in utils.get_clean_factor_and_forward_returns
 288    demeaned : bool
 289        Control how to build factor weights
 290        -- see performance.factor_weights for a full explanation
 291    group_adjust : bool
 292        Control how to build factor weights
 293        -- see performance.factor_weights for a full explanation
 294    equal_weight : bool, optional
 295        Control how to build factor weights
 296        -- see performance.factor_weights for a full explanation
 297    by_asset: bool, optional
 298        If True, returns are reported separately for each asset.
 299
 300    Returns
 301    -------
 302    returns : pd.DataFrame
 303        Period wise factor returns
 304    """
 305
 306    date_idx = factor_data.index.names.index("date")
 307    freq = factor_data.index.levels[date_idx].freq
 308
 309    weights = factor_weights(factor_data, demeaned, group_adjust, equal_weight)
 310
 311    s = utils.get_forward_returns_columns(factor_data.columns)
 312    weighted_returns = factor_data[s].multiply(weights, axis=0)
 313
 314    if by_asset:
 315        returns = weighted_returns
 316    else:
 317        # requires at least one weighted return
 318        # otherwise returns np.nan
 319        returns = _reindex_to_freq(
 320            weighted_returns.groupby(level="date").sum(min_count=1), freq
 321        )
 322
 323    return returns

 326def factor_alpha_beta(
 327    factor_data: pd.DataFrame,
 328    returns: Optional[Union[pd.DataFrame, pd.Series]] = None,
 329    demeaned: bool = True,
 330    group_adjust: bool = False,
 331    equal_weight: bool = False,
 332) -> pd.DataFrame:
 333    """
 334    Compute the alpha (excess returns), alpha t-stat (alpha significance),
 335    and beta (market exposure) of a factor. A regression is run with
 336    the period wise factor universe mean return as the independent variable
 337    and mean period wise return from a portfolio weighted by factor values
 338    as the dependent variable.
 339
 340    Parameters
 341    ----------
 342    factor_data : pd.DataFrame - MultiIndex
 343        A MultiIndex DataFrame indexed by date (level 0) and asset (level 1),
 344        containing the values for a single alpha factor, forward returns for
 345        each period, the factor quantile/bin that factor value belongs to, and
 346        (optionally) the group the asset belongs to.
 347        - See full explanation in utils.get_clean_factor_and_forward_returns
 348    returns : pd.DataFrame, optional
 349        Period wise factor returns. If this is None then it will be computed
 350        with 'factor_returns' function and the passed flags: 'demeaned',
 351        'group_adjust', 'equal_weight'
 352    demeaned : bool
 353        Control how to build factor returns used for alpha/beta computation
 354        -- see performance.factor_return for a full explanation
 355    group_adjust : bool
 356        Control how to build factor returns used for alpha/beta computation
 357        -- see performance.factor_return for a full explanation
 358    equal_weight : bool, optional
 359        Control how to build factor returns used for alpha/beta computation
 360        -- see performance.factor_return for a full explanation
 361
 362    Returns
 363    -------
 364    alpha_beta : pd.Series
 365        A list containing the alpha, beta, a t-stat(alpha)
 366        for the given factor and forward returns.
 367    """
 368
 369    if returns is None:
 370        returns = factor_returns(factor_data, demeaned, group_adjust, equal_weight)
 371
 372    universe_ret = (
 373        factor_data.groupby(level="date")[
 374            utils.get_forward_returns_columns(factor_data.columns)
 375        ]
 376        .mean()
 377        .reindex(returns.index, axis=0)
 378    )
 379
 380    if isinstance(returns, pd.Series):
 381        returns.name = universe_ret.columns.values[0]
 382        returns = pd.DataFrame(returns)
 383
 384    alpha_beta = pd.DataFrame()
 385    for period in returns.columns.values:
 386        x = universe_ret[period].values
 387        y = returns[period].values
 388        x = add_constant(x)
 389
 390        reg_fit = OLS(y, x, missing="drop").fit()
 391        try:
 392            alpha, beta = reg_fit.params
 393        except ValueError:
 394            alpha_beta.loc["Ann. alpha", period] = np.nan
 395            alpha_beta.loc["beta", period] = np.nan
 396        else:
 397            freq_adjust = utils._coerce_timedelta_days(252) / utils._coerce_timedelta(
 398                str(period)
 399            )
 400
 401            alpha_beta.loc["Ann. alpha", period] = (1 + alpha) ** freq_adjust - 1
 402            alpha_beta.loc["beta", period] = beta
 403
 404    return alpha_beta

 429def positions(weights, period, freq=None):
 430    """
 431    Builds net position values time series, the portfolio percentage invested
 432    in each position.
 433
 434    Parameters
 435    ----------
 436    weights: pd.Series
 437        pd.Series containing factor weights, the index contains timestamps at
 438        which the trades are computed and the values correspond to assets
 439        weights
 440        - see factor_weights for more details
 441    period: pandas.Timedelta or string
 442        Assets holding period (1 day, 2 mins, 3 hours etc). It can be a
 443        Timedelta or a string in the format accepted by Timedelta constructor
 444        ('1 days', '1D', '30m', '3h', '1D1h', etc)
 445    freq : pandas DateOffset, optional
 446        Used to specify a particular trading calendar. If not present
 447        weights.index.freq will be used
 448
 449    Returns
 450    -------
 451    pd.DataFrame
 452        Assets positions series, datetime on index, assets on columns.
 453        Example:
 454            index                 'AAPL'         'MSFT'          cash
 455            2004-01-09 10:30:00   13939.3800     -14012.9930     711.5585
 456            2004-01-09 15:30:00       0.00       -16012.9930     411.5585
 457            2004-01-12 10:30:00   14492.6300     -14624.8700       0.0
 458            2004-01-12 15:30:00   14874.5400     -15841.2500       0.0
 459            2004-01-13 10:30:00   -13853.2800    13653.6400      -43.6375
 460    """
 461
 462    weights = weights.unstack()
 463
 464    if not isinstance(period, pd.Timedelta):
 465        period = utils._coerce_timedelta(period)
 466
 467    if freq is None:
 468        freq = weights.index.freq
 469
 470    if freq is None:
 471        freq = BDay()
 472        warnings.warn("'freq' not set, using business day calendar", UserWarning)
 473
 474    #
 475    # weights index contains factor computation timestamps, then add returns
 476    # timestamps too (factor timestamps + period) and save them to 'full_idx'
 477    # 'full_idx' index will contain an entry for each point in time the weights
 478    # change and hence they have to be re-computed
 479    #
 480    trades_idx = weights.index.copy()
 481    returns_idx = utils.add_custom_calendar_timedelta(trades_idx, period, freq)
 482    weights_idx = trades_idx.union(returns_idx)
 483    # Remove duplicated dates when trading dates include weekend timestamps.
 484    weights_idx = weights_idx.drop_duplicates()
 485
 486    #
 487    # Compute portfolio weights for each point in time contained in the index
 488    #
 489    portfolio_weights = pd.DataFrame(
 490        index=weights_idx, columns=weights.columns, dtype=float
 491    )
 492    active_weights = []
 493
 494    for curr_time in weights_idx:
 495
 496        #
 497        # fetch new weights that become available at curr_time and store them
 498        # in active weights
 499        #
 500        if curr_time in weights.index:
 501            assets_weights = weights.loc[curr_time]
 502            expire_ts = utils.add_custom_calendar_timedelta(curr_time, period, freq)
 503            active_weights.append((expire_ts, assets_weights))
 504
 505        #
 506        # remove expired entry in active_weights (older than 'period')
 507        #
 508        active_weights = [
 509            (expiry, allocation)
 510            for expiry, allocation in active_weights
 511            if expiry > curr_time
 512        ]
 513
 514        if not active_weights:
 515            continue
 516        #
 517        # Compute total weights for curr_time and store them
 518        #
 519        tot_weights = [w for (ts, w) in active_weights]
 520        tot_weights = pd.concat(tot_weights, axis=1)
 521        tot_weights = tot_weights.sum(axis=1)
 522        tot_weights /= tot_weights.abs().sum()
 523
 524        portfolio_weights.loc[curr_time] = tot_weights
 525
 526    return portfolio_weights.fillna(0)

 529def mean_return_by_quantile(
 530    factor_data,
 531    by_date=False,
 532    by_group=False,
 533    demeaned=True,
 534    group_adjust=False,
 535):
 536    """
 537    Computes mean returns for factor quantiles across
 538    provided forward returns columns.
 539
 540    Parameters
 541    ----------
 542    factor_data : pd.DataFrame - MultiIndex
 543        A MultiIndex DataFrame indexed by date (level 0) and asset (level 1),
 544        containing the values for a single alpha factor, forward returns for
 545        each period, the factor quantile/bin that factor value belongs to, and
 546        (optionally) the group the asset belongs to.
 547        - See full explanation in utils.get_clean_factor_and_forward_returns
 548    by_date : bool
 549        If True, compute quantile bucket returns separately for each date.
 550    by_group : bool
 551        If True, compute quantile bucket returns separately for each group. Defaults to False unless the factor data
 552        has the 'group' column.
 553    demeaned : bool
 554        Compute demeaned mean returns across the entire universe per date (long short portfolio). Defaults to True.
 555    group_adjust : bool
 556        Returns demeaning will occur on the group level. When this flag is True, forward return demeaning occurs, at
 557        group level, regardless the value of 'demean' flag. Defaults to False unless the factor data has the 'group'
 558        column.
 559
 560    Returns
 561    -------
 562    mean_ret : pd.DataFrame
 563        Mean period wise returns by specified factor quantile.
 564    std_error_ret : pd.DataFrame
 565        Standard error of returns by specified quantile.
 566    """
 567
 568    if group_adjust:
 569        if "group" not in factor_data.columns:
 570            raise ValueError("group_adjust=True requires a 'group' column in factor_data")
 571        grouper = [factor_data.index.get_level_values("date")] + ["group"]
 572        factor_data = utils.demean_forward_returns(factor_data, grouper)
 573    elif demeaned:
 574        factor_data = utils.demean_forward_returns(factor_data)
 575    else:
 576        factor_data = factor_data.copy()
 577
 578    if "factor_quantile" not in factor_data.columns:
 579        raise ValueError("factor_data must contain a 'factor_quantile' column")
 580    grouper = ["factor_quantile", factor_data.index.get_level_values("date")]
 581
 582    if by_group:
 583        if "group" not in factor_data.columns:
 584            raise ValueError("by_group=True requires a 'group' column in factor_data")
 585        grouper.append("group")
 586
 587    group_stats = factor_data.groupby(grouper, observed=True)[
 588        utils.get_forward_returns_columns(factor_data.columns)
 589    ].agg(["mean", "std", "count"])
 590
 591    mean_ret = group_stats.T.xs("mean", level=1).T
 592
 593    if not by_date:
 594        grouper = [mean_ret.index.get_level_values("factor_quantile")]
 595        if by_group:
 596            grouper.append(mean_ret.index.get_level_values("group"))
 597        group_stats = mean_ret.groupby(grouper, observed=True).agg(["mean", "std", "count"])
 598        mean_ret = group_stats.T.xs("mean", level=1).T
 599
 600    std_error_ret = group_stats.T.xs("std", level=1).T / np.sqrt(
 601        group_stats.T.xs("count", level=1).T
 602    )
 603
 604    return mean_ret, std_error_ret

 607def compute_mean_returns_spread(mean_returns, upper_quant, lower_quant, std_err=None):
 608    """
 609    Computes the difference between the mean returns of
 610    two quantiles. Optionally, computes the standard error
 611    of this difference.
 612
 613    Parameters
 614    ----------
 615    mean_returns : pd.DataFrame
 616        DataFrame of mean period wise returns by quantile.
 617        MultiIndex containing date and quantile.
 618        See mean_return_by_quantile.
 619    upper_quant : int
 620        Quantile of mean return from which we
 621        wish to subtract lower quantile mean return.
 622    lower_quant : int
 623        Quantile of mean return we wish to subtract
 624        from upper quantile mean return.
 625    std_err : pd.DataFrame, optional
 626        Period wise standard error in mean return by quantile.
 627        Takes the same form as mean_returns.
 628
 629    Returns
 630    -------
 631    mean_return_difference : pd.Series
 632        Period wise difference in quantile returns.
 633    joint_std_err : pd.Series
 634        Period wise standard error of the difference in quantile returns.
 635        if std_err is None, this will be None
 636    """
 637
 638    mean_return_difference = mean_returns.xs(
 639        upper_quant, level="factor_quantile"
 640    ) - mean_returns.xs(lower_quant, level="factor_quantile")
 641    # Report spread at 100% gross leverage (50% long, 50% short).
 642    mean_return_difference = mean_return_difference / 2.0
 643
 644    if std_err is None:
 645        joint_std_err = None
 646    else:
 647        std1 = std_err.xs(upper_quant, level="factor_quantile")
 648        std2 = std_err.xs(lower_quant, level="factor_quantile")
 649        joint_std_err = np.sqrt(std1**2 + std2**2) / 2.0
 650
 651    return mean_return_difference, joint_std_err

 655    """
 656    Computes the proportion of names in a factor quantile that were
 657    not in that quantile in the previous period.
 658
 659    Parameters
 660    ----------
 661    quantile_factor : pd.Series
 662        DataFrame with date, asset and factor quantile.
 663    quantile : int
 664        Quantile on which to perform turnover analysis.
 665    period: int, optional
 666        Number of days over which to calculate the turnover.
 667
 668    Returns
 669    -------
 670    quant_turnover : pd.Series
 671        Period by period turnover for that quantile.
 672    """
 673
 674    # keep DateTimeIndex frequency information
 675    date_idx = quantile_factor.index.names.index("date")
 676    freq = getattr(quantile_factor.index.levels[date_idx], "freq", None)
 677
 678    quant_names = quantile_factor[quantile_factor == quantile]
 679    if freq is not None:
 680        quant_name_sets = (
 681            quant_names.groupby(level=["date"])
 682            .apply(lambda x: set(x.index.get_level_values("asset")))
 683            .asfreq(freq)
 684        )
 685    else:
 686        quant_name_sets = (
 687            quant_names.groupby(level=["date"])
 688            .apply(lambda x: set(x.index.get_level_values("asset")))
 689        )
 690
 691    name_shifted = quant_name_sets.shift(periods=period)
 692
 693    new_names = (quant_name_sets - name_shifted).dropna()
 694
 695    def f(xs):
 696        return 0 if pd.isna(xs) else len(xs)
 697
 698    def g(xs):
 699        return 1 if pd.isna(xs) else len(xs)
 700
 701    quant_turnover = (new_names.apply(f) / quant_name_sets.apply(g)).rename(quantile)
 702    return quant_turnover

 705def factor_rank_autocorrelation(factor_data, period=1):
 706    """
 707    Computes autocorrelation of mean factor ranks in specified time spans.
 708    We must compare period to period factor ranks rather than factor values
 709    to account for systematic shifts in the factor values of all names or names
 710    within a group. This metric is useful for measuring the turnover of a
 711    factor. If the value of a factor for each name changes randomly from period
 712    to period, we'd expect an autocorrelation of 0.
 713
 714    Parameters
 715    ----------
 716    factor_data : pd.DataFrame - MultiIndex
 717        A MultiIndex DataFrame indexed by date (level 0) and asset (level 1),
 718        containing the values for a single alpha factor, forward returns for
 719        each period, the factor quantile/bin that factor value belongs to, and
 720        (optionally) the group the asset belongs to.
 721        - See full explanation in utils.get_clean_factor_and_forward_returns
 722    period: int, optional
 723        Number of days over which to calculate the turnover.
 724
 725    Returns
 726    -------
 727    autocorr : pd.Series
 728        Rolling 1 period (defined by time_rule) autocorrelation of
 729        factor values.
 730    """
 731    # grouper = [factor_data.index.get_level_values('date')]
 732
 733    date_idx = factor_data.index.names.index("date")
 734    freq = getattr(factor_data.index.levels[date_idx], "freq", None)
 735
 736    if freq is not None:
 737        asset_ranks_by_day = (
 738            factor_data.groupby(level="date")["factor"]
 739            .rank()
 740            .reset_index()
 741            .pivot(index="date", columns="asset", values="factor")
 742            .asfreq(freq)
 743        )
 744    else:
 745        asset_ranks_by_day = (
 746            factor_data.groupby(level="date")["factor"]
 747            .rank()
 748            .reset_index()
 749            .pivot(index="date", columns="asset", values="factor")
 750        )
 751
 752    # asset_factor_rank = ranks
 753
 754    asset_shifted = asset_ranks_by_day.shift(period)
 755
 756    return (
 757        asset_ranks_by_day.corrwith(asset_shifted, axis=1).rename(period) if freq is None else asset_ranks_by_day.corrwith(asset_shifted, axis=1).rename(period).asfreq(freq)
 758    )

 761def common_start_returns(
 762    factor,
 763    returns,
 764    before,
 765    after,
 766    cumulative=False,
 767    mean_by_date=False,
 768    demean_by=None,
 769):
 770    """
 771    A date and equity pair is extracted from each index row in the factor
 772    dataframe and for each of these pairs a return series is built starting
 773    from 'before' the date and ending 'after' the date specified in the pair.
 774    All those returns series are then aligned to a common index (-before to
 775    after) and returned as a single DataFrame
 776
 777    Parameters
 778    ----------
 779    factor : pd.DataFrame
 780        DataFrame with at least date and equity as index, the columns are
 781        irrelevant
 782    returns : pd.DataFrame
 783        A wide form Pandas DataFrame indexed by date with assets in the
 784        columns. Returns data should span the factor analysis time period
 785        plus/minus an additional buffer window corresponding to after/before
 786        period parameters.
 787    before:
 788        How many returns to load before factor date
 789    after:
 790        How many returns to load after factor date
 791    cumulative: bool, optional
 792        Whether the given returns are cumulative. If False the given
 793        returns are assumed to be daily.
 794    mean_by_date: bool, optional
 795        If True, compute mean returns for each date and return that
 796        instead of a return series for each asset
 797    demean_by: pd.DataFrame, optional
 798        DataFrame with at least date and equity as index, the columns are
 799        irrelevant. For each date a list of equities is extracted from
 800        'demean_by' index and used as universe to compute demeaned mean
 801        returns (long short portfolio)
 802
 803    Returns
 804    -------
 805    aligned_returns : pd.DataFrame
 806        Dataframe containing returns series for each factor aligned to the same
 807        index: -before to after
 808    """
 809    if not cumulative:
 810        returns = returns.apply(cumulative_returns, axis=0)
 811
 812    all_returns = []
 813
 814    for timestamp, df in factor.groupby(level="date"):
 815
 816        equities = df.index.get_level_values("asset")
 817
 818        try:
 819            day_zero_index = returns.index.get_loc(timestamp)
 820        except KeyError:
 821            continue
 822
 823        starting_index = max(day_zero_index - before, 0)
 824        ending_index = min(day_zero_index + after + 1, len(returns.index))
 825
 826        equities_slice = set(equities)
 827        if demean_by is not None:
 828            demean_equities = demean_by.loc[timestamp].index.get_level_values("asset")
 829            equities_slice |= set(demean_equities)
 830
 831        series = returns.loc[
 832            returns.index[starting_index:ending_index], list(equities_slice)
 833        ]
 834        series.index = range(
 835            starting_index - day_zero_index, ending_index - day_zero_index
 836        )
 837
 838        if demean_by is not None:
 839            mean = series.loc[:, demean_equities].mean(axis=1)
 840            series = series.loc[:, equities]
 841            series = series.sub(mean, axis=0)
 842
 843        if mean_by_date:
 844            series = series.mean(axis=1)
 845
 846        all_returns.append(series)
 847
 848    return pd.concat(all_returns, axis=1)

 851def average_cumulative_return_by_quantile(
 852    factor_data,
 853    returns,
 854    periods_before=10,
 855    periods_after=15,
 856    demeaned=True,
 857    group_adjust=False,
 858    by_group=False,
 859):
 860    """
 861    Plots average cumulative returns by factor quantiles in the period range
 862    defined by -periods_before to periods_after
 863
 864    Parameters
 865    ----------
 866    factor_data : pd.DataFrame - MultiIndex
 867        A MultiIndex DataFrame indexed by date (level 0) and asset (level 1),
 868        containing the values for a single alpha factor, forward returns for
 869        each period, the factor quantile/bin that factor value belongs to, and
 870        (optionally) the group the asset belongs to.
 871        - See full explanation in utils.get_clean_factor_and_forward_returns
 872    returns : pd.DataFrame
 873        A wide form Pandas DataFrame indexed by date with assets in the
 874        columns. Returns data should span the factor analysis time period
 875        plus/minus an additional buffer window corresponding to periods_after/
 876        periods_before parameters.
 877    periods_before : int, optional
 878        How many periods before factor to plot
 879    periods_after  : int, optional
 880        How many periods after factor to plot
 881    demeaned : bool, optional
 882        Compute demeaned mean returns (long short portfolio)
 883    group_adjust : bool
 884        Returns demeaning will occur on the group level (group
 885        neutral portfolio)
 886    by_group : bool
 887        If True, compute cumulative returns separately for each group
 888
 889    Returns
 890    -------
 891    cumulative returns and std deviation : pd.DataFrame
 892        A MultiIndex DataFrame indexed by quantile (level 0) and mean/std
 893        (level 1) and the values on the columns in range from
 894        -periods_before to periods_after
 895        If by_group=True the index will have an additional 'group' level
 896        ::
 897            ---------------------------------------------------
 898                        |       | -2  | -1  |  0  |  1  | ...
 899            ---------------------------------------------------
 900              quantile  |       |     |     |     |     |
 901            ---------------------------------------------------
 902                        | mean  |  x  |  x  |  x  |  x  |
 903                 1      ---------------------------------------
 904                        | std   |  x  |  x  |  x  |  x  |
 905            ---------------------------------------------------
 906                        | mean  |  x  |  x  |  x  |  x  |
 907                 2      ---------------------------------------
 908                        | std   |  x  |  x  |  x  |  x  |
 909            ---------------------------------------------------
 910                ...     |                 ...
 911            ---------------------------------------------------
 912    """
 913
 914    def cumulative_return_around_event(q_fact, demean_by):
 915        return common_start_returns(
 916            q_fact,
 917            returns,
 918            periods_before,
 919            periods_after,
 920            cumulative=True,
 921            mean_by_date=True,
 922            demean_by=demean_by,
 923        )
 924
 925    def average_cumulative_return(q_fact, demean_by):
 926        q_returns = cumulative_return_around_event(q_fact, demean_by)
 927        q_returns.replace([np.inf, -np.inf], np.nan, inplace=True)
 928
 929        return pd.DataFrame(
 930            {
 931                "mean": q_returns.mean(skipna=True, axis=1),
 932                "std": q_returns.std(skipna=True, axis=1),
 933            }
 934        ).T
 935
 936    if by_group:
 937        #
 938        # Compute quantile cumulative returns separately for each group
 939        # Deman those returns accordingly to 'group_adjust' and 'demeaned'
 940        #
 941        returns_bygroup = []
 942
 943        for group, g_data in factor_data.groupby("group", observed=True):
 944            g_fq = g_data["factor_quantile"]
 945            if group_adjust:
 946                demean_by = g_fq  # demeans at group level
 947            elif demeaned:
 948                demean_by = factor_data["factor_quantile"]  # demean by all
 949            else:
 950                demean_by = None
 951            #
 952            # Align cumulative return from different dates to the same index
 953            # then compute mean and std
 954            #
 955            avgcumret = g_fq.groupby(g_fq).apply(average_cumulative_return, demean_by)
 956            if len(avgcumret) == 0:
 957                continue
 958
 959            avgcumret["group"] = group
 960            avgcumret.set_index("group", append=True, inplace=True)
 961            returns_bygroup.append(avgcumret)
 962
 963        return pd.concat(returns_bygroup, axis=0)
 964
 965    else:
 966        #
 967        # Compute quantile cumulative returns for the full factor_data
 968        # Align cumulative return from different dates to the same index
 969        # then compute mean and std
 970        # Deman those returns accordingly to 'group_adjust' and 'demeaned'
 971        #
 972        if group_adjust:
 973            all_returns = []
 974            for group, g_data in factor_data.groupby("group", observed=True):
 975                g_fq = g_data["factor_quantile"]
 976                avgcumret = g_fq.groupby(g_fq).apply(
 977                    cumulative_return_around_event, g_fq
 978                )
 979                all_returns.append(avgcumret)
 980            q_returns = pd.concat(all_returns, axis=1)
 981            q_returns = pd.DataFrame(
 982                {"mean": q_returns.mean(axis=1), "std": q_returns.std(axis=1)}
 983            )
 984            q_returns = q_returns.unstack(level=1)
 985            try:
 986                return q_returns.stack(level=0, future_stack=True)
 987            except TypeError:
 988                return q_returns.stack(level=0)
 989        elif demeaned:
 990            fq = factor_data["factor_quantile"]
 991            return fq.groupby(fq).apply(average_cumulative_return, fq)
 992        else:
 993            fq = factor_data["factor_quantile"]
 994            return fq.groupby(fq).apply(average_cumulative_return, None)

1143def create_pyfolio_input(
1144    factor_data,
1145    period,
1146    capital=None,
1147    long_short=True,
1148    group_neutral=False,
1149    equal_weight=False,
1150    quantiles=None,
1151    groups=None,
1152    benchmark_period="1D",
1153):
1154    """
1155    Simulate a portfolio using the input factor and returns the portfolio
1156    performance data properly formatted for Pyfolio analysis.
1157
1158    For more details on how this portfolio is built see:
1159    - performance.cumulative_returns (how the portfolio returns are computed)
1160    - performance.factor_weights (how assets weights are computed)
1161
1162    Parameters
1163    ----------
1164    factor_data : pd.DataFrame - MultiIndex
1165        A MultiIndex DataFrame indexed by date (level 0) and asset (level 1),
1166        containing the values for a single alpha factor, forward returns for
1167        each period, the factor quantile/bin that factor value belongs to,
1168        and (optionally) the group the asset belongs to.
1169        - See full explanation in utils.get_clean_factor_and_forward_returns
1170    period : string
1171        'factor_data' column name corresponding to the 'period' returns to be
1172        used in the computation of porfolio returns
1173    capital : float, optional
1174        If set, then compute 'positions' in dollar amount instead of percentage
1175    long_short : bool, optional
1176        if True enforce a dollar neutral long-short portfolio: asset weights
1177        will be computed by demeaning factor values and dividing by the sum of
1178        their absolute value (achieving gross leverage of 1) which will cause
1179        the portfolio to hold both long and short positions and the total
1180        weights of both long and short positions will be equal.
1181        If False the portfolio weights will be computed dividing the factor
1182        values and  by the sum of their absolute value (achieving gross
1183        leverage of 1). Positive factor values will generate long positions and
1184        negative factor values will produce short positions so that a factor
1185        with only posive values will result in a long only portfolio.
1186    group_neutral : bool, optional
1187        If True simulates a group neutral portfolio: the portfolio weights
1188        will be computed so that each group will weigh the same.
1189        if 'long_short' is enabled the factor values demeaning will occur on
1190        the group level resulting in a dollar neutral, group neutral,
1191        long-short portfolio.
1192        If False group information will not be used in weights computation.
1193    equal_weight : bool, optional
1194        if True the assets will be equal-weighted. If long_short is True then
1195        the factor universe will be split in two equal sized groups with the
1196        top assets in long positions and bottom assets in short positions.
1197        if False the assets will be factor-weighed, see 'long_short' argument
1198    quantiles: sequence[int], optional
1199        Use only specific quantiles in the computation. By default all
1200        quantiles are used
1201    groups: sequence[string], optional
1202        Use only specific groups in the computation. By default all groups
1203        are used
1204    benchmark_period : string, optional
1205        By default benchmark returns are computed as the factor universe mean
1206        daily returns but 'benchmark_period' allows to choose a 'factor_data'
1207        column corresponding to the returns to be used in the computation of
1208        benchmark returns. More generally benchmark returns are computed as the
1209        factor universe returns traded at 'benchmark_period' frequency, equal
1210        weighting and long only
1211
1212
1213    Returns
1214    -------
1215     returns : pd.Series
1216        Daily returns of the strategy, noncumulative.
1217         - Time series with decimal returns.
1218         - Example:
1219            2015-07-16    -0.012143
1220            2015-07-17    0.045350
1221            2015-07-20    0.030957
1222            2015-07-21    0.004902
1223
1224     positions : pd.DataFrame
1225        Time series of dollar amount (or percentage when 'capital' is not
1226        provided) invested in each position and cash.
1227         - Days where stocks are not held can be represented by 0.
1228         - Non-working capital is labelled 'cash'
1229         - Example:
1230            index         'AAPL'         'MSFT'          cash
1231            2004-01-09    13939.3800     -14012.9930     711.5585
1232            2004-01-12    14492.6300     -14624.8700     27.1821
1233            2004-01-13    -13853.2800    13653.6400      -43.6375
1234
1235
1236     benchmark : pd.Series
1237        Benchmark returns computed as the factor universe mean daily returns.
1238
1239    """
1240
1241    #
1242    # Build returns:
1243    # we don't know the frequency at which the factor returns are computed but
1244    # pyfolio wants daily returns. So we compute the cumulative returns of the
1245    # factor, then resample it at 1 day frequency and finally compute daily
1246    # returns
1247    #
1248    cumrets = factor_cumulative_returns(
1249        factor_data,
1250        period,
1251        long_short,
1252        group_neutral,
1253        equal_weight,
1254        quantiles,
1255        groups,
1256    )
1257    cumrets = cumrets.resample("1D").last().ffill()
1258    returns = cumrets.pct_change().fillna(0)
1259
1260    #
1261    # Build positions. As pyfolio asks for daily position we have to resample
1262    # the positions returned by 'factor_positions' at 1 day frequency and
1263    # recompute the weights so that the sum of daily weights is 1.0
1264    #
1265    positions = factor_positions(
1266        factor_data,
1267        period,
1268        long_short,
1269        group_neutral,
1270        equal_weight,
1271        quantiles,
1272        groups,
1273    )
1274    positions = positions.resample("1D").sum().ffill()
1275    positions = positions.div(positions.abs().sum(axis=1), axis=0).fillna(0)
1276    positions["cash"] = 1.0 - positions.sum(axis=1)
1277
1278    # transform percentage positions to dollar positions
1279    if capital is not None:
1280        positions = positions.mul(cumrets.reindex(positions.index) * capital, axis=0)
1281
1282    #
1283    #
1284    #
1285    # Build benchmark returns as the factor universe mean returns traded at

204        weights are computed by demeaning factor values and dividing by the sum
205        of their absolute value (achieving gross leverage of 1). The sum of
206        positive weights will be the same as the negative weights (absolute
207        value), suitable for a dollar neutral long-short portfolio. Defaults to True.
208    group_adjust : bool
